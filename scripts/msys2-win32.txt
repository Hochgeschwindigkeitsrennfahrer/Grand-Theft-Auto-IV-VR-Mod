# Meson cross-file for MSYS2 MINGW32 (i686).
# IronWolf build-win32.txt expects i686-w64-mingw32-ar — under MSYS2 that tool is often just "ar".

cpp = 'i686-w64-mingw32-g++'
ar = 'ar'
strip = 'strip'
windres = 'windres'

[properties]
needs_exe_wrapper = true

[built-in options]
c_args = ['-O3', '-march=i686']
cpp_args = ['-O3', '-march=i686']
c_link_args = ['-s']
cpp_link_args = ['-s']

[host_machine]
system = 'windows'
cpu_family = 'x86'
cpu = 'x86'
endian = 'little'
